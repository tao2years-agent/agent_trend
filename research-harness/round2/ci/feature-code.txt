def get_user_occurrences_status_query(account_id, number):
    return [
        {
            "$match": {
                "_number": int(number),
                "$or": [
                    {"owner._account_id": account_id},
                    {"reviewer_updates.reviewer._account_id": account_id},
                ],
            }
        },
        {
            "$project": {
                "owner": {
                    "$cond": {
                        "if": {"$eq": ["$owner._account_id", account_id]},
                        "then": "$owner",
                        "else": "$$REMOVE",
                    }
                },
                "reviewer_updates": {
                    "$filter": {
                        "input": "$reviewer_updates",
                        "as": "review",
                        "cond": {"$eq": ["$$review.reviewer._account_id", account_id]},
                    }
                },
            }
        },
    ]

def get_bots_success_failure_interval_diff(bots, key, recheck_number):
    bots_diff_avg_success_sequence = []
    bots_diff_avg_failure_sequence = []

    for bot_id, bot_info in bots.items():
        if bot_id != "":
            bot_diff_avg_sequence = fetch_bot_diff_avg_sequence(
                bot_id,
                key,
                recheck_number,
                bot_info["started_at"],
            )

            bots_diff_avg_success_sequence.append(
                bot_diff_avg_sequence["diff_avg_success_interval"]
            )
            bots_diff_avg_failure_sequence.append(
                bot_diff_avg_sequence["diff_avg_failure_interval"]
            )

    return max(bots_diff_avg_success_sequence), min(bots_diff_avg_failure_sequence)


def get_bots_success_ratio(bots, feature_key):
    bots_success_ratios = []
    bots_flip_success_ratios = []
    jobs_success_ratios = []
    for bot_id, bot_info in bots.items():
        if bot_id != "":
            bot_flips = fetch_bot_flips(feature_key, bot_info["started_at"], bot_id)
            all_flips = (
                bot_flips["flip_from_failure_to_success_num"]
                + bot_flips["flip_from_failure_to_failure_num"]
            )
            current_bot_flip_success_ratio = (
                bot_flips["flip_from_failure_to_success_num"] / all_flips
                if all_flips
                else DEFAULT_SUCCESS_RATIO
            )
            bots_flip_success_ratios.append(current_bot_flip_success_ratio)

            bot_occurrences = fetch_bot_occurrences(
                feature_key, bot_info["started_at"], bot_id
            )
            current_bot_success_ratio = (
                bot_occurrences["num_success_occurences"]
                / bot_occurrences["num_all_occurences"]
                if bot_occurrences["num_all_occurences"]
                else DEFAULT_SUCCESS_RATIO
            )
            bots_success_ratios.append(current_bot_success_ratio)

            for job in bot_info["jobs"]:
                if job["voting"]:
                    job_occurrences = fetch_job_occurrences(
                        feature_key, bot_info["started_at"], bot_id, job["name"]
                    )
                    current_job_success_ratio = (
                        job_occurrences["num_success_occurences"]
                        / job_occurrences["num_all_occurences"]
                        if job_occurrences["num_all_occurences"]
                        else DEFAULT_SUCCESS_RATIO
                    )
                    jobs_success_ratios.append(current_job_success_ratio)

    bots_success_ratio = (
        min(bots_success_ratios) if bots_success_ratios else DEFAULT_SUCCESS_RATIO
    )
    jobs_success_ratio = (
        min(jobs_success_ratios) if jobs_success_ratios else DEFAULT_SUCCESS_RATIO
    )
    bots_flip_success_ratio = (
        min(bots_flip_success_ratios)
        if bots_flip_success_ratios
        else DEFAULT_SUCCESS_RATIO
    )

    return (
        bots_success_ratio,
        jobs_success_ratio,
        bots_flip_success_ratio,
    )


def get_user_success_ratio(user, key):
    user_occurrences = fetch_user_occurrences(
        user["account_id"], user["created_at"], key
    )
    user_success_ratio = (
        user_occurrences["num_success_occurences"]
        / user_occurrences["num_all_occurences"]
        if user_occurrences["num_all_occurences"]
        else DEFAULT_SUCCESS_RATIO
    )
    return user_success_ratio


def get_user_status(account_id, number):
    status = 0
    user_occurrences_status_query = get_user_occurrences_status_query(
        account_id, number
    )
    user_occurrences_status = list(
        change_info_full_collection.aggregate(user_occurrences_status_query)
    )

    if user_occurrences_status:
        user_occurrences_status = user_occurrences_status[0]
        if "owner" in user_occurrences_status:
            status = 2
        elif user_occurrences_status["reviewer_updates"]:
            status = 1

    return status

def get_patch_info(revision, project, feature_key):
    lines_inserted = 0
    lines_deleted = 0
    numbers_of_unique_files = 0
    files_occurrences = []

    for filename, file_info in revision["files"].items():
        numbers_of_unique_files += 1
        lines_inserted += file_info.get("lines_inserted", 0)
        lines_deleted += file_info.get("lines_deleted", 0)

        file_occurrences = fetch_file_occurrences(f"{project}/{filename}", feature_key)
        file_occurrences_ratio = (
            file_occurrences["num_of_success_occurrences"]
            / file_occurrences["num_of_occurrences"]
            if file_occurrences["num_of_occurrences"] != 0
            else DEFAULT_SUCCESS_RATIO
        )
        files_occurrences.append(file_occurrences_ratio)

    files_success_ratio = (
        min(files_occurrences) if files_occurrences else DEFAULT_SUCCESS_RATIO
    )
    return lines_inserted, lines_deleted, numbers_of_unique_files, files_success_ratio

def calculate_features(recheck_builds):
    features = []

    for recheck_build in tqdm(recheck_builds):
        key = ast.literal_eval(recheck_build["key"])  # (change_id, patch_number)
        number = str(key[0])

        len_bots = len(recheck_build["bot"])
        if len_bots > 1:
            for index, bot in enumerate(recheck_build["bot"]):
                recheck_number = bot["recheck_number"]
                user = next(
                    user
                    for user in recheck_build["rerun_requestors"]
                    if user["recheck_number"] == int(recheck_number)
                )
                user_full_info = author_info_collection.find_one(
                    {"_account_id": int(user["account_id"])}
                )
                if (
                    index != 0
                    and user_full_info["is_bot"] == False
                    and recheck_build["bot"][index - 1]["status"] == "failure"
                ):
                    feature_key = (*key, int(recheck_number))

                    # bots_success_ratio
                    new_bots = {
                        k: v
                        for k, v in bot.items()
                        if k not in ["recheck_number", "status"]
                    }
                    diff_avg_success, diff_avg_failure = (
                        get_bots_success_failure_interval_diff(
                            new_bots, key, recheck_number
                        )
                    )
                    bots_success_ratio, jobs_success_ratio, bots_flip_success_ratio = (
                        get_bots_success_ratio(new_bots, feature_key)
                    )

                    # user_success_ratio

                    user_success_ratio = get_user_success_ratio(user, key)
                    user_experience = get_user_experience(user, number)
                    user_status = get_user_status(user["account_id"], number)

                    revision_info = revisions_collection.find_one(
                        {"number": int(number), "revision_number": int(key[1])}
                    )

                    project = revision_info["project"]
                    (
                        lines_inserted,
                        lines_deleted,
                        numbers_of_unique_files,
                        files_success_ratio,
                    ) = get_patch_info(revision_info, project, feature_key)

                    date_of_recheck = next(
                        user["created_at"]
                        for user in recheck_build["rerun_requestors"]
                        if str(user["recheck_number"]) == str(recheck_number)
                    )
                    date_object = datetime.datetime.strptime(
                        date_of_recheck, "%Y-%m-%d %H:%M:%S"
                    )

                    # Extract date and time features
                    month_number = date_object.month
                    day_of_week_number = date_object.weekday()
                    hour = date_object.hour
                    minute = date_object.minute

                    features.append(
                        {
                            "key": feature_key,
                            "user_status": user_status,
                            "user_success_ratio": user_success_ratio,
                            "bots_success_ratio": bots_success_ratio,
                            "bots_flip_success_ratio": bots_flip_success_ratio,
                            "jobs_success_ratio": jobs_success_ratio,
                            "lines_inserted": lines_inserted,
                            "lines_deleted": lines_deleted,
                            "numbers_of_unique_files": numbers_of_unique_files,
                            "files_success_ratio": files_success_ratio,
                            "user_experience": user_experience,
                            "queued_month": month_number,
                            "queued_day": day_of_week_number,
                            "queued_hour": hour,
                            "queued_minute": minute,
                            "date_of_recheck": date_of_recheck,
                            "diff_avg_success": diff_avg_success,
                            "diff_avg_failure": diff_avg_failure,
                            "project": project,
                            "label": (1 if bot["status"] == "success" else 0),
                        }
                    )
    return features


def divide_array(arr, num_pieces):
    k, m = divmod(len(arr), num_pieces)
    return [
        arr[i * k + min(i, m) : (i + 1) * k + min(i + 1, m)] for i in range(num_pieces)
    ]
